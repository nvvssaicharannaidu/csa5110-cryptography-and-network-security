import numpy as np

K = np.array([[9, 4], [5, 7]])
text = "MEETMEATTHEUSUALPLACEATTENRATHERTHANEIGHTOCLOCK".replace(" ", "")

if len(text) % 2:
    text += "X"

# Encrypt
cipher = ""
for i in range(0, len(text), 2):
    p = np.array([[ord(text[i])-65], [ord(text[i+1])-65]])
    c = K.dot(p) % 26
    cipher += chr(c[0,0]+65) + chr(c[1,0]+65)

print("Ciphertext:", cipher)

# Decrypt: K inverse modulo 26 = [[5,12],[11,25]]
Kinv = np.array([[5, 12], [11, 25]])
plain = ""

for i in range(0, len(cipher), 2):
    c = np.array([[ord(cipher[i])-65], [ord(cipher[i+1])-65]])
    p = Kinv.dot(c) % 26
    plain += chr(p[0,0]+65) + chr(p[1,0]+65)

print("Decrypted:", plain)
