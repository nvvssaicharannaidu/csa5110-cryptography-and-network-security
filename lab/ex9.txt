# Playfair Cipher Decryption

ciphertext = """KXJEY UREBE ZWEHE WRYTU HEYFS
KREHE GOYFI WTTTU OLKSY CAJPO
BOTEI ZONTX BYBNT GONEY CUZWR
GDSON SXBOU YWRHE BAAHY USEDQ"""

key = "MONARCHY"

# Create 5x5 Playfair matrix
alphabet = "ABCDEFGHIKLMNOPQRSTUVWXYZ"
key = "".join(dict.fromkeys(key.upper().replace("J", "I")))

matrix = key
for ch in alphabet:
    if ch not in matrix:
        matrix += ch

print("Playfair Matrix:")
for i in range(0, 25, 5):
    print(matrix[i:i+5])

# Decryption function
def decrypt_pair(a, b):
    r1, c1 = divmod(matrix.index(a), 5)
    r2, c2 = divmod(matrix.index(b), 5)

    if r1 == r2:
        return matrix[r1*5 + (c1-1)%5] + matrix[r2*5 + (c2-1)%5]
    elif c1 == c2:
        return matrix[((r1-1)%5)*5 + c1] + matrix[((r2-1)%5)*5 + c2]
    else:
        return matrix[r1*5 + c2] + matrix[r2*5 + c1]

# Remove spaces
ciphertext = ciphertext.replace(" ", "").replace("\n", "")

# Decrypt
plaintext = ""
for i in range(0, len(ciphertext), 2):
    plaintext += decrypt_pair(ciphertext[i], ciphertext[i+1])

print("\nDecrypted Message:")
print(plaintext)
